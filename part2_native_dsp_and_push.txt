# FILE 2: Patch lib/entry_music_data.dart for Automatic Daily Entry List Rotation
# ==============================================================================
python3 - << 'PYEOF'
with open("lib/entry_music_data.dart", "r") as f:
    code = f.read()

old_seed_logic = """      final List<String> pool = categorySeedQueries[categoryKey] ?? categorySeedQueries['ALL_1000']!;
      if (loadNextBatch) {
        _batchCursor = (_batchCursor + 4) % pool.length;
      }
      for (int i = 0; i < pool.length; i++) {
        queries.add(pool[(_batchCursor + i) % pool.length]);
      }"""

new_seed_logic = """      final List<String> pool = categorySeedQueries[categoryKey] ?? categorySeedQueries['ALL_1000']!;
      final DateTime now = DateTime.now();
      final int dailyOffset = (now.year * 365 + now.month * 31 + now.day) % pool.length;
      if (loadNextBatch) {
        _batchCursor = (_batchCursor + 4) % pool.length;
      } else {
        _batchCursor = dailyOffset;
      }
      for (int i = 0; i < pool.length; i++) {
        queries.add(pool[(_batchCursor + i) % pool.length]);
      }"""

if old_seed_logic in code:
    code = code.replace(old_seed_logic, new_seed_logic)

old_entry_sort = "out.sort((a, b) => ((b['playCount'] ?? 0) as int).compareTo((a['playCount'] ?? 0) as int));"
new_entry_sort = """final int dailySalt = DateTime.now().day * 17 + DateTime.now().month * 31;
    out.sort((a, b) {
      final int hashA = (a['id'].toString().hashCode ^ dailySalt) & 0x7fffffff;
      final int hashB = (b['id'].toString().hashCode ^ dailySalt) & 0x7fffffff;
      return hashB.compareTo(hashA);
    });"""

if old_entry_sort in code:
    code = code.replace(old_entry_sort, new_entry_sort)

with open("lib/entry_music_data.dart", "w") as f:
    f.write(code)
print("✅ Patched Page 4 Entry Music to rotate dynamically every day!")
PYEOF

cat << 'EOF' > .github/workflows/main.yml

        run: |
          MAIN_KT=$(find android/app/src/main -name "MainActivity.kt" | head -n 1)

            PKG_LINE=$(head -n 1 "$MAIN_KT")
            cat << 'KTEOF' > "$MAIN_KT"
          $PKG_LINE
          import android.media.MediaRecorder
          import android.media.audiofx.AcousticEchoCanceler
          import android.media.audiofx.AutomaticGainControl
          import android.media.audiofx.Equalizer
          import android.media.audiofx.NoiseSuppressor
          import android.os.Bundle
          import android.speech.tts.TextToSpeech
          import android.speech.tts.Voice
          import android.view.KeyEvent
          import io.flutter.embedding.android.FlutterActivity
          import io.flutter.embedding.engine.FlutterEngine
          import io.flutter.plugin.common.MethodChannel
          import java.io.File
          import java.util.Locale

          class MainActivity: FlutterActivity(), TextToSpeech.OnInitListener {
              private var tts: TextToSpeech? = null
              private var ttsReady: Boolean = false
              private var recorder: MediaRecorder? = null
              private var currentRecordPath: String? = null
              private var karaokeEq: Equalizer? = null
              private var methodChannel: MethodChannel? = null

              override fun onCreate(savedInstanceState: Bundle?) {
                  super.onCreate(savedInstanceState)
                  tts = TextToSpeech(this, this)
              }

              override fun onInit(status: Int) {
                  ttsReady = (status == TextToSpeech.SUCCESS)
              }

              override fun onKeyDown(keyCode: Int, event: KeyEvent?): Boolean {
                  if (keyCode == KeyEvent.KEYCODE_VOLUME_UP) {
                      methodChannel?.invokeMethod("onVolumeKey", "UP")
                  } else if (keyCode == KeyEvent.KEYCODE_VOLUME_DOWN) {
                      methodChannel?.invokeMethod("onVolumeKey", "DOWN")
                  }
                  return super.onKeyDown(keyCode, event)
              }

              private fun selectBestNaturalVoice(locale: Locale) {
                  try {
                      val voices = tts?.voices ?: return
                      var bestVoice: Voice? = null
                      for (v in voices) {
                          if (v.locale.language == locale.language) {
                              if (bestVoice == null || v.quality > bestVoice.quality || v.name.contains("network", true) || v.name.contains("wavenet", true)) {
                                  bestVoice = v
                              }
                          }
                      }
                      if (bestVoice != null) {
                          tts?.voice = bestVoice
                      } else {
                          tts?.language = locale
                      }
                  } catch (_: Exception) {
                      tts?.language = locale
                  }
              }

              private fun applyKaraokeVocalCutEq(enabled: Boolean) {
                  try {
                      if (karaokeEq == null) {
                          karaokeEq = Equalizer(0, 0)
                      }
                      karaokeEq?.enabled = enabled
                      if (enabled && karaokeEq != null) {
                          val bands = karaokeEq!!.numberOfBands.toInt()
                          val range = karaokeEq!!.bandLevelRange
                          val minLvl = range[0].toInt()
                          val maxLvl = range[1].toInt()
                          for (b in 0 until bands) {
                              val freqHz = karaokeEq!!.getCenterFreq(b.toShort()) / 1000
                              if (freqHz in 350..3800) {
                                  // Attenuate lead vocal center frequencies to turn normal song into Karaoke
                                  karaokeEq!!.setBandLevel(b.toShort(), (minLvl * 75 / 100).toShort())
                              } else if (freqHz < 220) {
                                  // Boost warm bass backing
                                  karaokeEq!!.setBandLevel(b.toShort(), (maxLvl * 55 / 100).toShort())
                              } else {
                                  // Crisp high percussion
                                  karaokeEq!!.setBandLevel(b.toShort(), (maxLvl * 35 / 100).toShort())
                              }
                          }
                      }
                  } catch (_: Exception) {}
              }

              private fun startIsolatedVoiceRecorder(fileName: String, useEchoCancelledSource: Boolean): String? {
                  try {
                      recorder?.release()
                      val outFile = File(filesDir, fileName + ".m4a")
                      currentRecordPath = outFile.absolutePath
                      recorder = MediaRecorder().apply {
                          // VOICE_COMMUNICATION activates Android Hardware AcousticEchoCanceler & NoiseSuppressor
                          // so playing Karaoke backing music is subtracted from the mic and ONLY user singing is recorded!
                          if (useEchoCancelledSource) {
                              setAudioSource(MediaRecorder.AudioSource.VOICE_COMMUNICATION)
                          } else {
                              setAudioSource(MediaRecorder.AudioSource.MIC)
                          }
                          setOutputFormat(MediaRecorder.OutputFormat.MPEG_4)
                          setAudioEncoder(MediaRecorder.AudioEncoder.AAC)
                          setAudioEncodingBitRate(192000)
                          setAudioSamplingRate(44100)
                          setOutputFile(currentRecordPath)
                          prepare()
                          start()
                      }
                      return currentRecordPath
                  } catch (_: Exception) {
                      return try {
                          recorder?.release()
                          val outFile = File(filesDir, fileName + ".m4a")
                          currentRecordPath = outFile.absolutePath
                          recorder = MediaRecorder().apply {
                              setAudioSource(MediaRecorder.AudioSource.MIC)
                              setOutputFormat(MediaRecorder.OutputFormat.MPEG_4)
                              setAudioEncoder(MediaRecorder.AudioEncoder.AAC)
                              setAudioEncodingBitRate(192000)
                              setAudioSamplingRate(44100)
                              setOutputFile(currentRecordPath)
                              prepare()
                              start()
                          }
                          currentRecordPath
                      } catch (_: Exception) {
                          null
                      }
                  }
              }

              override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
                  super.configureFlutterEngine(flutterEngine)
                  val ch = MethodChannel(flutterEngine.dartExecutor.binaryMessenger, "dj_native_tts")
                  methodChannel = ch
                  ch.setMethodCallHandler { call, result ->
                      when (call.method) {
                          "speak" -> {
                              if (!ttsReady || tts == null) {
                                  result.success(false)
                                  return@setMethodCallHandler
                              }
                              val text = call.argument<String>("text") ?: ""
                              val lang = call.argument<String>("lang") ?: "kn-IN"
                              val pitch = (call.argument<Double>("pitch") ?: 1.0).toFloat()
                              val rate = (call.argument<Double>("rate") ?: 0.98).toFloat()
                              val locale = Locale.forLanguageTag(lang)
                              val avail = tts?.isLanguageAvailable(locale) ?: TextToSpeech.LANG_NOT_SUPPORTED
                              if (avail < TextToSpeech.LANG_AVAILABLE) {
                                  result.success(false)
                                  return@setMethodCallHandler
                              }
                              tts?.stop()
                              selectBestNaturalVoice(locale)
                              tts?.setPitch(pitch)
                              tts?.setSpeechRate(rate)
                              val speakRes = tts?.speak(text, TextToSpeech.QUEUE_FLUSH, null, "dj_single_utt")
                              result.success(speakRes == TextToSpeech.SUCCESS)
                          }
                          "stopSpeak" -> {
                              tts?.stop()
                              result.success(true)
                          }
                          "setKaraokeVocalCut" -> {
                              val enabled = call.argument<Boolean>("enabled") ?: false
                              applyKaraokeVocalCutEq(enabled)
                              result.success(true)
                          }
                          "startCleanVocalRecord" -> {
                              val name = call.argument<String>("name") ?: "karaoke_vocal"
                              val path = startIsolatedVoiceRecorder(name, true)
                              result.success(path)
                          }
                          "stopCleanVocalRecord" -> {
                              try {
                                  recorder?.stop()
                                  recorder?.release()
                                  recorder = null
                                  result.success(currentRecordPath)
                              } catch (_: Exception) {
                                  recorder?.release()
                                  recorder = null
                                  result.success(currentRecordPath)
                              }
                          }
                          "startRecord" -> {
                              val name = call.argument<String>("name") ?: "voice_clip"
                              val path = startIsolatedVoiceRecorder(name, false)
                              result.success(path)
                          }
                          "stopRecord" -> {
                              try {
                                  recorder?.stop()
                                  recorder?.release()
                                  recorder = null
                                  result.success(currentRecordPath)
                              } catch (_: Exception) {
                                  recorder?.release()
                                  recorder = null
                                  result.success(currentRecordPath)
                              }
                          }
                          else -> result.notImplemented()
                      }
                  }
              }

              override fun onDestroy() {
                  tts?.stop()
                  tts?.shutdown()
                  recorder?.release()
                  karaokeEq?.release()
                  super.onDestroy()
              }
          }

              sed -i 's/<application/<application android:usesCleartextTraffic="true"/' "$MANIFEST"

              sed -i 's/<application/<uses-permission android:name="android.permission.INTERNET"\/>\n    <uses-permission android:name="android.permission.RECORD_AUDIO"\/>\n    <uses-permission android:name="android.permission.MODIFY_AUDIO_SETTINGS"\/>\n    <application/' "$MANIFEST"
            fi

                cp -f assets/icon.png "$dir/ic_launcher.png" || true

EOF
