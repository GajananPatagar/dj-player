  static Uint8List _buildWavBytes(String type) {
    const int sampleRate = 22050;
    final double durationSec =
        (type == 'CROWD' || type == 'SIX' || type == 'DHOL' || type == 'HEARTBEAT' || type == 'RISER')
            ? 1.15
            : 0.48;
    final int numSamples = (sampleRate * durationSec).toInt();
    final int dataSize = numSamples * 2;
    final ByteData bd = ByteData(44 + dataSize);

    final header = [
      0x52, 0x49, 0x46, 0x46,
      ..._int32Le(36 + dataSize),
      0x57, 0x41, 0x56, 0x45,
      0x66, 0x6D, 0x74, 0x20,
      16, 0, 0, 0, 1, 0, 1, 0,
      ..._int32Le(sampleRate),
      ..._int32Le(sampleRate * 2),
      2, 0, 16, 0,
      0x64, 0x61, 0x74, 0x61,
      ..._int32Le(dataSize),
    ];
    for (int i = 0; i < header.length; i++) {
      bd.setUint8(i, header[i]);
    }

    final Random rng = Random();
    for (int i = 0; i < numSamples; i++) {
      final double t = i / sampleRate;
      final double env = (1.0 - (i / numSamples)).clamp(0.0, 1.0);
      double sample = 0.0;

      if (type == 'BOWLED') {
        sample = (sin(2 * pi * (175 - t * 220) * t) * 0.6 + (rng.nextDouble() * 2 - 1) * 0.35) * env;
      } else if (type == 'NO_BALL') {
        final freq = 460 + 280 * sin(2 * pi * 6 * t);
        sample = sin(2 * pi * freq * t) * 0.65 * env;
      } else if (type == 'AIRHORN') {
        sample = (sin(2 * pi * 390 * t) + 0.6 * sin(2 * pi * 780 * t)) * 0.48 * env;
      } else if (type == 'LASER') {
        final freq = 1400 - (t * 2400);
        sample = sin(2 * pi * freq.clamp(100, 1500) * t) * 0.65 * env;
      } else if (type == 'SUB_DROP') {
        final freq = (130 - t * 95).clamp(38.0, 130.0);
        sample = sin(2 * pi * freq * t) * 0.88 * env;
      } else if (type == 'DHOL') {
        final pulse = (sin(2 * pi * 6.5 * t) > 0) ? 1.0 : 0.25;
        sample = sin(2 * pi * 115 * t) * pulse * 0.78 * env;
      } else if (type == 'TRUMPET') {
        final f = (t < 0.2) ? 523.25 : ((t < 0.4) ? 659.25 : 783.99);
        sample = (sin(2 * pi * f * t) + 0.4 * sin(4 * pi * f * t)) * 0.55 * env;
      } else if (type == 'HEARTBEAT') {
        final thump = ((t % 0.45) < 0.12) ? 1.0 : 0.05;
        sample = sin(2 * pi * 62 * t) * thump * 0.9;
      } else if (type == 'BELL') {
        sample = (sin(2 * pi * 880 * t) + 0.5 * sin(2 * pi * 1760 * t)) * 0.5 * env;
      } else if (type == 'RISER') {
        final freq = 120 + (t * 950);
        sample = sin(2 * pi * freq * t) * (t / durationSec).clamp(0.1, 0.85);
      } else if (type == 'SUB_TEST_50HZ') {
        sample = sin(2 * pi * 50 * t) * 0.85;
      } else if (type == 'TOP_TEST_1KHZ') {
        sample = sin(2 * pi * 1000 * t) * 0.55;
      } else {
        final crack = (t < 0.045) ? sin(2 * pi * 520 * t) * 0.85 : 0.0;
        final roar = (rng.nextDouble() * 2 - 1) * 0.28 * env;
        sample = (crack + roar).clamp(-1.0, 1.0);
      }

      final int pcm = (sample.clamp(-1.0, 1.0) * 32760).toInt();
      bd.setInt16(44 + i * 2, pcm, Endian.little);
    }
    return bd.buffer.asUint8List();
  }

  static List<int> _int32Le(int v) => [v & 0xff, (v >> 8) & 0xff, (v >> 16) & 0xff, (v >> 24) & 0xff];

  void _loadSquads() {
    final parsedA = _squadACtrl.text.split(',').map((e) => e.trim()).where((e) => e.isNotEmpty).toSet().toList();
    final parsedB = _squadBCtrl.text.split(',').map((e) => e.trim()).where((e) => e.isNotEmpty).toSet().toList();
    _squadA = parsedA.isEmpty ? ['Player 1', 'Player 2'] : parsedA;
    _squadB = parsedB.isEmpty ? ['Bowler 1', 'Keeper 1', 'Fielder 1'] : parsedB;

    if (!_squadA.contains(_striker)) _striker = _squadA[0];
    if (!_squadA.contains(_nonStriker)) _nonStriker = _squadA.length > 1 ? _squadA[1] : _squadA[0];
    if (!_squadB.contains(_wk)) _wk = _squadB[0];
    if (!_squadB.contains(_bowler)) _bowler = _squadB.length > 1 ? _squadB[1] : _squadB[0];
    if (!_squadB.contains(_fielder)) _fielder = _squadB.length > 2 ? _squadB[2] : _squadB[0];
    if (mounted) setState(() {});
  }

  String _oversText() => (_legalBalls ~/ 6).toString() + '.' + (_legalBalls % 6).toString();

  String _getBestBatterCap() {
    if (_batterRuns.isEmpty) return _striker + ' (0)';
    String best = _striker;
    int maxR = -1;
    _batterRuns.forEach((k, v) {
      if (v > maxR) {
        maxR = v;
        best = k;
      }
    });
    return best + ' (' + maxR.toString() + ')';
  }

  String _getBestBowlerCap() {
    if (_bowlerWickets.isEmpty) return _bowler + ' (0W)';
    String best = _bowler;
    int maxW = -1;
    _bowlerWickets.forEach((k, v) {
      if (v > maxW) {
        maxW = v;
        best = k;
      }
    });
    return best + ' (' + maxW.toString() + 'W)';
  }

  void _triggerBall(String eventType) {
    _saveUndoSnapshot();

    bool legal = true;
    bool rotate = false;
    bool isWicket = false;
    int addedRuns = 0;
    String badge = '0';
    String? incomingBatterName;

    if (_freeHit &&
        (eventType == 'BOWLED' || eventType == 'CAUGHT' || eventType == 'CAUGHT_WK' || eventType == 'STUMPED' || eventType == 'LBW')) {
      eventType = 'DOT';
    }

    switch (eventType) {
      case 'DOT':
        badge = '0';
        _consecutiveWickets = 0;
        break;
      case 'RUN_1':
        addedRuns = 1;
        rotate = true;
        badge = '1';
        _consecutiveWickets = 0;
        break;
      case 'RUN_2':
        addedRuns = 2;
        badge = '2';
        _consecutiveWickets = 0;
        break;
      case 'RUN_3':
        addedRuns = 3;
        rotate = true;
        badge = '3';
        _consecutiveWickets = 0;
        break;
      case 'FOUR':
        addedRuns = 4;
        badge = '4';
        _consecutiveWickets = 0;
        break;
      case 'SIX':
        addedRuns = 6;
        badge = '6';
        _consecutiveWickets = 0;
        break;
      case 'BOWLED':
      case 'CAUGHT':
      case 'CAUGHT_WK':
      case 'RUN_OUT':
      case 'STUMPED':
      case 'LBW':
        _wickets += 1;
        _consecutiveWickets += 1;
        isWicket = true;
        badge = 'W';
        break;
      case 'NO_BALL':
        _runs += 1;
        _bowlerRuns[_bowler] = (_bowlerRuns[_bowler] ?? 0) + 1;
        legal = false;
        _freeHit = true;
        badge = 'NB';
        break;
      case 'WIDE':
        _runs += 1;
        _bowlerRuns[_bowler] = (_bowlerRuns[_bowler] ?? 0) + 1;
        legal = false;
        badge = 'WD';
        break;
    }

    if (addedRuns > 0) {
      _runs += addedRuns;
      _partnershipRuns += addedRuns;
      _batterRuns[_striker] = (_batterRuns[_striker] ?? 0) + addedRuns;
      _bowlerRuns[_bowler] = (_bowlerRuns[_bowler] ?? 0) + addedRuns;
    }

    if (legal) {
      _legalBalls += 1;
      _partnershipBalls += 1;
      _batterBalls[_striker] = (_batterBalls[_striker] ?? 0) + 1;
      if (eventType != 'NO_BALL') _freeHit = false;
    }

    final line = _generateNonRepeatingLine(eventType);

    setState(() {
      _thisOver.add(badge);
      if (rotate) {
        final t = _striker;
        _striker = _nonStriker;
        _nonStriker = t;
      }
      if (isWicket && _squadA.isNotEmpty) {
        _bowlerWickets[_bowler] = (_bowlerWickets[_bowler] ?? 0) + 1;
        _fallOfWickets.add(_runs.toString() + '/' + _wickets.toString() + ' (' + _striker + ')');
        _partnershipRuns = 0;
        _partnershipBalls = 0;
        final nextIdx = (_wickets + 1) % _squadA.length;
        incomingBatterName = _squadA[nextIdx];
        _striker = incomingBatterName!;
      }
      if (legal && _legalBalls % 6 == 0) {
        _thisOver.clear();
        final t = _striker;
        _striker = _nonStriker;
        _nonStriker = t;
      }
    });

    final String banter = _buildCoCommentatorBanter(eventType, incomingBatterName);
    setState(() {
      _feed.insert(
        0,
        '[' + _oversText() + '] ' + line + (banter.isNotEmpty ? ('\n🎙 Co-Comm: ' + banter) : ''),
      );
    });

    AppSoundEngine.playSynthesizedSfx(eventType);
    if ((isWicket && _autoWalkoutBgmOnWicket) || eventType == 'SIX' || eventType == 'FOUR') {
      MasterVolumeController.autoCricketStingerTrigger.value =
          eventType + '_' + DateTime.now().millisecondsSinceEpoch.toString();
    }
    AppSoundEngine.speakRealVoice(line, _commLanguage, eventKey: eventType, coCommentatorText: banter);
  }
